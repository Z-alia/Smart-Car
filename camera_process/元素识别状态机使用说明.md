## 🏗️ 系统架构

### 核心数据流

```
图像采集 → 扫线算法 → 元素识别 → 状态机处理 → 控制输出
    ↓          ↓          ↓           ↓           ↓
OV2640    lineinfo[]   Element    Element_    电机/舵机
                                recognition()   控制
```

### 状态机层次结构

```
┌─────────────────────────────────────────────────────────┐
│                   Element_recognition()                  │
│                   (主状态机调度函数)                      │
└─────────────────────────────────────────────────────────┘
                           │
        ┌──────────────────┼──────────────────┐
        ▼                  ▼                  ▼
    None状态          元素状态             default
 (enter_task)      (各元素处理)         (未知处理)
        │                  │
        └──────┬───────────┘
               ▼
        各元素子状态机
     (如:圆环、十字、坡道等)
```

## 🎯 元素类型定义

### Element_range 枚举类型

```c
typedef enum {
    None = 0,            // 无元素(正常巡线)
    Left_ring = 1,       // 左圆环
    Right_ring = 2,      // 右圆环
    Slope = 3,           // 坡道
    broken_circuit = 4,  // 断路
    obstacle = 5,        // 障碍物
    ingarage = 6,        // 入库
    outgarage = 7,       // 出库
    crossing = 8,        // 十字路口
    black_obstacle = 9,  // 小型黑色路障
    zebra = 10          // 斑马线
} Element_range;
```

```c
// 当前元素状态
```

## 🔄 状态机工作流程

### 1. 主状态机 - Element_recognition()

```c
void Element_recognition() {
    switch(Element) {
        case None:
            enter_task();  // 元素识别核心函数
            break;
            
        case Left_ring:
            // 左圆环处理子状态机
            left_ring_first_angle();      // 寻找第一个角点
            left_ring_circular_arc();     // 检测圆弧
            left_ring_begin_turn();       // 开始转弯
            left_ring_second_angle();     // 第二个角点
            left_ring_in_loop();          // 完全入环
            left_ring_prepare_out();      // 准备出环
            left_ring_out_loop_turn();    // 出环转弯
            left_ring_out_angle();        // 出环角点
            left_ring_out_loop();         // 直道出环
            left_ring_straight_out_angle();// 最后角点
            left_ring_complete_out();     // 完全出环
            break;
            
        case Right_ring:
            // 右圆环处理(类似左圆环)
            right_ring_first_angle();
            // ... 其他子状态
            break;
            
        case Slope:
            slope_down();     // 下坡检测
            slope_out();      // 出坡道
            break;
            
        case crossing:
            cross_running();  // 十字路口行驶
            cross_running2();
            cross_out();      // 退出十字
            break;
            
        case zebra:
            zebra_stop();     // 斑马线停车
            zebra_out();      // 退出斑马线
            break;
            
        // ... 其他元素处理
    }
}
```

### 2. 元素识别核心 - enter_task()

```c
void enter_task() {
    switch(Element_rem.Element_data[Element_rem.Element_count]) {
        case 0:  // 自动扫描模式
            if(mycar.RUNTIME >= setpara.begin_time) {
                left_ring_first_angle();   // 检测左圆环
                right_ring_first_angle();  // 检测右圆环
                slope_enter();             // 检测坡道
                cross_enter();             // 检测十字
                zebra_enter();             // 检测斑马线
                // ... 其他元素检测
            }
            break;
            
        case 1:  // 预设左圆环
            left_ring_first_angle();
            break;
            
        case 2:  // 预设右圆环
            right_ring_first_angle();
            break;
            
        // ... 其他预设元素
            
        default:
            out_element();  // 退出当前元素
            break;
    }
}
```

### 3. 进入元素 - enter_element()

```c
void enter_element(Element_range element) {

}
```

### 4. 退出元素 - out_element()

```c
void out_element() {

}

void clear_all_flags() {
    Element = None;                    // 恢复无元素状态
    
    // 清除积分器
    clear_angle_integeral();
    clear_distant_integeral();
    
    // 清除圆环标志
    watch.InLoop = 0;
    watch.OutLoop = 0;
    watch.InLoopAngleL = 120;
    // ... 其他标志位
    
    // 恢复PID参数
    mycar.speed_ctrl = 1;
    mycar.pid_ctrl = 1;
    change_pid_para(&CAM_Turn, &CAM_FUZZY_PID);
    
    // 清除其他元素标志
    watch.cross_flag = 0;
    watch.slope_flag = 0;
    watch.zebra_flag = 0;
    watch.garage_stop = 0;
    // ...
}
```

### main.c 中的使用

```c
#include "Element_recognition.h"
#include "scan_line.h"
#include "Binarization.h"

int main(void) {
    // 1. 系统初始化
    HAL_Init();
    SystemClock_Config();

                                      60, Display_Width, 10);

            // 4.2 元素识别与状态机处理 (核心!)

            // 4.3 控制输出
            // 根据 Element 状态和 watch 标志位进行控制
            // 例如: motor_control(), servo_control()
            
            // 4.4 显示
            draw_edge();
            show_ov2640_image_int8(0, 0, imo[0], 
                                  Display_Width, Display_Height, 
                                  Display_Width, Display_Height);
        }
    }
}
```

## ⚙️ 元素配置方法

### 元素初始化 - element_init()

```c
void element_init() {
    Element_rem.Element_count = 0;  // 元素计数器归零
    
    // 配置元素序列
    // 方式1: 左出库模式 (start_mode == 1 或 3)
    if (setpara.start_mode == 1 || setpara.start_mode == 3) {
        Element_rem.Element_data[0] = 7;  // 出库
        for (int i = 1; i < 21; i++) {
            Element_rem.Element_data[i] = setpara.set_element[i];
        }
    }
    
    // 方式2: 右出库模式 (start_mode == 2 或 4)
    else if (setpara.start_mode == 2 || setpara.start_mode == 4) {
        // 元素顺序镜像翻转
        // 左圆环 ↔ 右圆环
        for (int i = 1; i < 21; i++) {
            if (setpara.set_element[i] == 1)
                Element_rem.Element_data[count-i] = 2;  // 左→右
            else if (setpara.set_element[i] == 2)
                Element_rem.Element_data[count-i] = 1;  // 右→左
            else
                Element_rem.Element_data[count-i] = setpara.set_element[i];
        }
    }
    
    // 配置圆环类型序列
    Element_rem.loop_count = 0;
    Element_rem.loop_data[0] = setpara.loop_data[0];  // 小圆环/大圆环
    Element_rem.loop_data[1] = setpara.loop_data[1];
    // ...
}
```

### 参数配置示例

```c
// 在你的参数配置中
struct setpara_struct {
    uint8 start_mode;           // 启动模式 1=左出库 2=右出库

    uint32 begin_time;          // 开始识别元素的时间(ms)
    uint32 slope_begin_time;    // 开始识别坡道的时间
    uint32 zebra_begin_time;    // 开始识别斑马线的时间
    
    float loop_target_speed;    // 圆环目标速度
    float slope_speed;          // 坡道速度
    // ... 其他参数
};

// 示例配置
setpara.start_mode = 1;              // 左出库
setpara.set_element[0] = 7;          // 出库
setpara.set_element[1] = 1;          // 左圆环
setpara.set_element[2] = 8;          // 十字路口

setpara.loop_data[1] = 1;            // 第2个圆环:大圆环

setpara.begin_time = 2000;           // 2秒后开始识别
setpara.loop_target_speed = 1.5f;    // 圆环速度 1.5m/s
```

## 🎨 常见元素处理流程

### 1. 左圆环处理流程

```c
// 状态转换图:
// InLoop=0 → InLoop=1 → InLoop=2 → InLoop=3 → InLoop=4 → 
// InLoop=5 → InLoop=6 → OutLoop=1 → OutLoop=2 → 完成

// 第1步: 检测第一个角点 (InLoop=1)
void left_ring_first_angle() {
    if (watch.InLoop != 0 && watch.InLoop != 1) return;
    
    for (int y = 20; y < 100; y++) {
        if (/* 检测到左边线丢失 && 右边线完整 */) {
            watch.InLoopAngleL = y;  // 记录角点位置
            if (Element == None) {
                left_ring_confirm();  // 确认是否真的是圆环
            }
            break;
        }
    }
}

// 圆环确认
void left_ring_confirm() {
    // 检测右边线是否平滑(排除右圆环)
    // 检测左侧是否有足够白色区域
    if (/* 确认条件满足 */) {
        enter_element(Left_ring);    // 进入左圆环状态
        set_speed(setpara.loop_target_speed);  // 设置速度
        change_pid_para(&CAM_Turn, &setpara.loop_turn_PID);  // 切换PID
        watch.InLoop = 1;  // 设置子状态
    }
}

// 第2步: 检测圆弧 (InLoop=2)
void left_ring_circular_arc() {
    if (watch.InLoop != 1) return;
    
    if (/* 检测到弧形特征 */) {
        watch.InLoop = 2;
    }
}

// ... 其他步骤类似

// 完成出环
void left_ring_complete_out() {
    if (watch.OutLoop == 2) {
        if (/* 边线恢复正常 */) {
            out_element();  // 退出圆环,恢复正常巡线
        }
    }
}
```

### 2. 坡道处理流程

```c
// 坡道状态: slope_flag=0 → 1 → 2 → 3 → 0

// 第1步: 检测上坡
void slope_enter() {
    if (Element == None) {
        // 方法1: 激光测距检测
        if (dl1b_distance_mm < 1000) {
            slope_count++;
            if (slope_count >= 3) {
                set_speed(setpara.slope_speed);
                watch.slope_flag = 1;  // 准备上坡
            }
        }
        
        // 方法2: IMU姿态检测
        if (imu.pitch > 15 && mycar.RUNTIME > setpara.slope_begin_time) {
            enter_element(Slope);      // 进入坡道元素
            set_speed(setpara.slope_speed);
            watch.slope_flag = 2;      // 正在上坡
            watch.angle_far_line = 50; // 减少远端扫线
            begin_distant_integeral(10000);  // 开始距离积分
            beep2(4, 100);             // 蜂鸣器提示
        }
    }
}

// 第2步: 检测下坡
void slope_down() {
    if ((imu.pitch < 0 || get_integeral_state(&distance_integral) == 2) 
        && watch.slope_flag == 2) {
        watch.angle_far_line = setpara.far_line;  // 恢复扫线
        watch.slope_flag = 3;  // 正在下坡
        begin_distant_integeral(10000);
    }
}

// 第3步: 退出坡道
void slope_out() {
    if (watch.slope_flag == 3 && 
        get_integeral_state(&distance_integral) == 2) {
        imu.pitch = 0;
        out_element();  // 退出坡道元素
    }
}
```

### 3. 斑马线处理流程

```c
// 斑马线状态: zebra_flag=0 → 1 → 2 → 0

// 第1步: 检测斑马线
void zebra_enter() {
    if (mycar.RUNTIME < setpara.zebra_begin_time) return;
    
    if (Element == None) {
        int zebra_line_count = 0;
        
        // 统计斑马线行数
        for (int y = 20; y < 100; y++) {
            if (lineinfo[y].zebra_flag == 1) {
                zebra_line_count++;
            }
        }
        
        // 判断是否为斑马线
        if (zebra_line_count > 10) {
            enter_element(zebra);  // 进入斑马线元素
            set_speed(0);          // 停车
            watch.zebra_flag = 1;
        }
    }
}

// 第2步: 停车
void zebra_stop() {
    if (watch.zebra_flag == 1) {
        set_speed(0);  // 持续停车
        watch.stop_count++;
        
        if (watch.stop_count >= 100) {  // 停车一定时间
            watch.zebra_flag = 2;       // 准备启动
        }
    }
}

// 第3步: 退出斑马线
void zebra_out() {
    if (watch.zebra_flag == 2 && mycar.present_speed <= 0) {
        set_speed(setpara.speed_min);  // 恢复速度
        
        if (/* 已离开斑马线区域 */) {
            out_element();  // 退出斑马线元素
        }
    }
}
```

## ⚠️ 注意事项

### 1. 调用顺序

```c
// ✅ 正确顺序
Binarization();         // 先二值化
scan_line();            // 再扫线
Element_recognition();  // 最后元素识别

// ❌ 错误: 没有扫线就识别元素
Element_recognition();  // 错误! lineinfo[]未更新
```

### 2. 状态清除

```c
// 退出元素时必须清除所有标志
void custom_element_out() {
    out_element();  // 自动清除所有标志
    
    // 不要手动清除,会导致遗漏
    // watch.InLoop = 0;  // 不推荐
}
```

### 3. 元素冲突

```c
// 同一时刻只能有一个元素
if (Element != None) return;  // 已在处理其他元素,不检测新元素

// 或使用优先级
if (Element == None || Element == zebra) {
    // 某些元素可以被高优先级元素打断
}
```

### 4. 时间限制

```c
// 使用运行时间避免误识别
if (mycar.RUNTIME < setpara.begin_time) return;  // 启动后一段时间才开始识别
```

## 📊 元素状态流转图

```
启动
  ↓
element_init() ← 初始化元素序列
  ↓
Element = None (正常巡线)
  ↓
enter_task() ← 扫描识别元素
  ↓
检测到元素?
  │
  ├─ Yes → enter_element(xxx) → Element = xxx
  │           ↓
  │         各元素子状态机处理
  │           ↓
  │         完成条件满足?
  │           ↓
  │         out_element() → Element = None
  │           ↓
  └─ No ──→ 继续巡线 ──┘
```
