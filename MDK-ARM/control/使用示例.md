## 🚀 快速开始

### 方法1: 使用默认参数(最简单)

```c
#include "control.h"

void main(void)
{
    // 硬件初始化...
    
    // 使用默认参数初始化串级PID(推荐新手)
    control_init_cascade_pid_default();
    
    while(1) {
        float vL = get_left_speed();
        float vR = get_right_speed();
        float pwm_L, pwm_R;
        
        control_loop(1.0f, vL, vR, &pwm_L, &pwm_R);
        
        set_motor_pwm(pwm_L, pwm_R);
        delay_ms(1);
    }
}
```

## 🔥 在线热调参

### 示例1: 按键调参(串级PID)

```c
#include "control.h"

// 全局参数,方便按键修改
float g_speed_ratio = 10.0f;
float g_max_diff_speed = 1.5f;
float g_smooth_alpha = 0.20f;

void main(void)
{
    control_init_cascade_pid_default();
    
    while(1) {
        // 按键检测
        if (key_pressed(KEY_UP)) {
            g_speed_ratio += 0.5f;  // 增大转向强度
            control_update_cascade_outer(g_speed_ratio, g_max_diff_speed, g_smooth_alpha);
            printf("speed_ratio: %.2f\n", g_speed_ratio);
        }
        
        if (key_pressed(KEY_DOWN)) {
            g_speed_ratio -= 0.5f;  // 减小转向强度
            control_update_cascade_outer(g_speed_ratio, g_max_diff_speed, g_smooth_alpha);
            printf("speed_ratio: %.2f\n", g_speed_ratio);
        }
        
        if (key_pressed(KEY_LEFT)) {
            g_smooth_alpha -= 0.05f;  // 更平滑
            g_smooth_alpha = clampf(g_smooth_alpha, 0.05f, 1.0f);
            control_update_cascade_outer(g_speed_ratio, g_max_diff_speed, g_smooth_alpha);
            printf("smooth_alpha: %.2f\n", g_smooth_alpha);
        }
        
        if (key_pressed(KEY_RIGHT)) {
            g_smooth_alpha += 0.05f;  // 更快响应
            g_smooth_alpha = clampf(g_smooth_alpha, 0.05f, 1.0f);
            control_update_cascade_outer(g_speed_ratio, g_max_diff_speed, g_smooth_alpha);
            printf("smooth_alpha: %.2f\n", g_smooth_alpha);
        }
        
        // 控制循环
        float vL = get_left_speed();
        float vR = get_right_speed();
        float pwm_L, pwm_R;
        
        control_loop(1.0f, vL, vR, &pwm_L, &pwm_R);
        
        set_motor_pwm(pwm_L, pwm_R);
        delay_ms(1);
    }
}
```

### 示例2: 串口调参(ADRC)

```c
// 全局ADRC参数
float g_adrc_b = 5.0f;
float g_adrc_w0 = 20.0f;
float g_adrc_r = 50.0f;
float g_adrc_kp = 1.0f;
float g_adrc_kd = 0.5f;

void uart_rx_callback(void)
{
    char cmd;
    float value;
    
    // 解析上位机命令: "B=5.0\n", "W=25.0\n" 等
    if (sscanf(uart_buffer, "%c=%f", &cmd, &value) == 2) {
        switch(cmd) {
            case 'B': // 修改 b 参数
                g_adrc_b = value;
                control_update_adrc_core(g_adrc_b, g_adrc_w0, g_adrc_r, 
                                        g_adrc_kp, g_adrc_kd);
                printf("ADRC b updated: %.2f\n", g_adrc_b);
                break;
                
            case 'W': // 修改 w0 参数
                g_adrc_w0 = value;
                control_update_adrc_core(g_adrc_b, g_adrc_w0, g_adrc_r, 
                                        g_adrc_kp, g_adrc_kd);
                printf("ADRC w0 updated: %.2f\n", g_adrc_w0);
                break;
                
            case 'R': // 修改 r 参数
                g_adrc_r = value;
                control_update_adrc_core(g_adrc_b, g_adrc_w0, g_adrc_r, 
                                        g_adrc_kp, g_adrc_kd);
                printf("ADRC r updated: %.2f\n", g_adrc_r);
                break;
                
            case 'P': // 修改 kp 参数
                g_adrc_kp = value;
                control_update_adrc_core(g_adrc_b, g_adrc_w0, g_adrc_r, 
                                        g_adrc_kp, g_adrc_kd);
                printf("ADRC kp updated: %.2f\n", g_adrc_kp);
                break;
                
            case 'D': // 修改 kd 参数
                g_adrc_kd = value;
                control_update_adrc_core(g_adrc_b, g_adrc_w0, g_adrc_r, 
                                        g_adrc_kp, g_adrc_kd);
                printf("ADRC kd updated: %.2f\n", g_adrc_kd);
                break;
        }
    }
}
```

### 示例3: 非线性增益开关

```c
int g_nonlinear_enable = 0;
float g_nonlinear_k = 1e-6f;

void toggle_nonlinear_gain(void)
{
    if (key_pressed(KEY_FUNC)) {
        g_nonlinear_enable = !g_nonlinear_enable;
        control_set_nonlinear_gain(g_nonlinear_enable, g_nonlinear_k);
        
        if (g_nonlinear_enable) {
            printf("Nonlinear Gain: ON (k=%.2e)\n", g_nonlinear_k);
        } else {
            printf("Nonlinear Gain: OFF\n");
        }
    }
    
    // 调节非线性系数
    if (g_nonlinear_enable && key_pressed(KEY_ADJUST)) {
        g_nonlinear_k *= 1.5f;  // 增大1.5倍
        if (g_nonlinear_k > 1e-4f) g_nonlinear_k = 1e-7f; // 循环
        control_set_nonlinear_gain(g_nonlinear_enable, g_nonlinear_k);
        printf("Nonlinear k: %.2e\n", g_nonlinear_k);
    }
}
```

```python
import serial
import tkinter as tk
from tkinter import ttk

class SmartCarTuner:
    def __init__(self):
        self.ser = serial.Serial('COM3', 115200)  # 根据实际端口修改
        
        # 创建GUI
        self.root = tk.Tk()
        self.root.title("Smart Car Parameter Tuner")
        
        # 串级PID参数
        ttk.Label(self.root, text="Speed Ratio:").grid(row=0, column=0)
        self.speed_ratio = ttk.Scale(self.root, from_=5.0, to=15.0, 
                                      orient='horizontal', length=200,
                                      command=self.update_speed_ratio)
        self.speed_ratio.set(10.0)
        self.speed_ratio.grid(row=0, column=1)
        
        ttk.Label(self.root, text="Smooth Alpha:").grid(row=1, column=0)
        self.smooth_alpha = ttk.Scale(self.root, from_=0.05, to=1.0, 
                                       orient='horizontal', length=200,
                                       command=self.update_smooth_alpha)
        self.smooth_alpha.set(0.20)
        self.smooth_alpha.grid(row=1, column=1)
        
        # ADRC参数
        ttk.Label(self.root, text="ADRC w0:").grid(row=2, column=0)
        self.adrc_w0 = ttk.Scale(self.root, from_=10.0, to=100.0, 
                                 orient='horizontal', length=200,
                                 command=self.update_adrc_w0)
        self.adrc_w0.set(20.0)
        self.adrc_w0.grid(row=2, column=1)
        
        self.root.mainloop()
    
    def update_speed_ratio(self, value):

        self.ser.write(cmd.encode())
    
    def update_smooth_alpha(self, value):
        cmd = f"SA={float(value):.3f}\n"
        self.ser.write(cmd.encode())
    
    def update_adrc_w0(self, value):
        cmd = f"W={float(value):.2f}\n"
        self.ser.write(cmd.encode())

if __name__ == '__main__':
    tuner = SmartCarTuner()
```

## 🎓 最佳实践

### 调参流程推荐

```c
void tuning_workflow(void)
{
    // Step 1: 使用默认参数初始化
    control_init_cascade_pid_default();
    
    // Step 2: 测量实际轮距,修改half_track
    control_update_cascade_outer(10.0f, 1.5f, 0.20f);
    // 或重新初始化:
    // CascadePIDConfig config = {...};
    // config.half_track = 0.070f;  // 实测14cm轮距
    // control_init_cascade_pid(&config);
    
    // Step 3: 调节speed_ratio(核心参数)
    // 从保守值开始,逐步增大
    control_update_cascade_outer(8.0f, 1.5f, 0.20f);  // 保守
    // 测试 → 偏离不回正 → 增大
    control_update_cascade_outer(10.0f, 1.5f, 0.20f); // 适中
    // 测试 → 左右晃动 → 减小
    
    // Step 4: 微调smooth_alpha
    // 高速时减小,低速时可增大
    control_update_cascade_outer(10.0f, 1.5f, 0.15f); // 高速平滑
    
    // Step 5: (可选)启用非线性增益
    control_set_nonlinear_gain(1, 1e-6f);
    
    // Step 6: 速度环PID微调(如果需要)
    control_update_cascade_speed_pid(12.0f, 0.6f, 0.15f,  // 左轮
                                     12.0f, 0.6f, 0.15f); // 右轮
}
```

## ⚠️ 注意事项

1. **参数范围**: 务必在推荐范围内调节,超出范围可能失控
2. **渐进调节**: 每次只调一个参数,改动幅度不超过20%
3. **重置状态**: 修改PID/ADRC参数后会自动重置状态,可能有短暂抖动
4. **保存参数**: 调好的参数要及时保存到配置文件或Flash
5. **安全测试**: 新参数先在安全环境测试,确认无误再上赛道
