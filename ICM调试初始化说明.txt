/**
 * ICM-42688P 调试初始化函数
 * 在 LCD 上显示每一步的执行状态，帮助定位阻塞位置
 */

// 在 ICM-426688P.c 中添加以下调试函数

#ifdef ICM_DEBUG_INIT

extern void LCD_DisplayString(uint16_t x, uint16_t y, char *str);

uint8_t ICM42688P_Init_Debug(void)
{
    uint8_t whoami = 0;
    
    LCD_DisplayString(10, 30, "Step 1: CS High");
    cs_high();
    delay_ms(20);
    
    LCD_DisplayString(10, 30, "Step 2: Reset  ");
    ICM42688P_Software_Reset();
    delay_ms(100);
    
    LCD_DisplayString(10, 30, "Step 3: Bank 0 ");
    ICM42688P_Bank_Select(0);
    delay_ms(10);
    
    LCD_DisplayString(10, 30, "Step 4: WHOAMI ");
    uint8_t retry = 0;
    for(retry = 0; retry < 3; retry++)
    {
        ICM42688P_ReadRegister(ICM42688P_WHOAMI, &whoami, 1);
        if (whoami == 0x47)
        {
            break;
        }
        delay_ms(20);
    }
    
    if (whoami != 0x47)
    {
        LCD_DisplayString(10, 30, "WHOAMI FAIL    ");
        char buf[20];
        sprintf(buf, "ID=0x%02X", whoami);
        LCD_DisplayString(10, 50, buf);
        return 1;
    }
    
    LCD_DisplayString(10, 30, "Step 5: Clock  ");
    ICM42688P_Clock_Config();
    delay_ms(10);
    
    LCD_DisplayString(10, 30, "Step 6: IRQ    ");
    ICM42688P_Interrupt_Config();
    delay_ms(10);
    
    LCD_DisplayString(10, 30, "Step 7: ODR    ");
    ICM42688P_ODR_Config();
    delay_ms(10);
    
    LCD_DisplayString(10, 30, "Step 8: Start  ");
    ICM42688P_Start();
    delay_ms(10);
    
    LCD_DisplayString(10, 30, "Init Success!  ");
    return 0;
}

#endif

/* 使用方法：
 * 1. 在 ICM-42688P.h 中添加：#define ICM_DEBUG_INIT
 * 2. 在 main.c 中调用：ICM42688P_Init_Debug();
 * 3. 观察 LCD 显示，看卡在哪一步
 * 4. 如果卡住，LCD 会停留在某个 "Step X" 的显示上
 */
